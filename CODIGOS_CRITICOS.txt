const agentOutput = $input.first().json.output;
const phoneNumber = $('Extraer Datos WhatsApp').first().json.phone_number;
const restaurantId = $('Vector Search Products').first().json.restaurant_id;

console.log('🛒 [Detectar Pedido] Analizando output del agente');

const isPedidoCompleto = agentOutput.includes('PEDIDO_COMPLETO');

let productos = [];
let outputLimpio = agentOutput;

if (isPedidoCompleto) {
  try {
    const lines = agentOutput.split('\n');
    let inItems = false;
    
    for (const line of lines) {
      if (line.includes('Itens:')) {
        inItems = true;
        continue;
      }
      if (line.includes('Fornecedor')) {
        inItems = false;
      }
      if (inItems && line.trim().startsWith('-')) {
        const match = line.match(/- (.+?):\s*([\d.]+)\s*(\w+)/);
        if (match) {
          productos.push({
            produto: match[1].trim(),
            quantidade: parseFloat(match[2]),
            unidade: match[3]
          });
        }
      }
    }
    
    console.log(`✅ [Detectar Pedido] ${productos.length} produtos detectados`);
    
    outputLimpio = agentOutput
      .replace(/PEDIDO_COMPLETO/g, '')
      .trim();
    
  } catch (error) {
    console.error('❌ [Detectar Pedido] Error parseando:', error);
  }
}

return [{
  json: {
    phone_number: phoneNumber,
    restaurant_id: restaurantId,
    pedido_completo: isPedidoCompleto,
    produtos: productos,
    output: outputLimpio,
    original_output: agentOutput
  }
}];

// 🔍 Validar disponibilidad de precios antes de procesar pedido
const productos = $input.first().json.produtos;
const restaurantId = $input.first().json.restaurant_id;
const phoneNumber = $input.first().json.phone_number;

console.log(`🔍 [Validar Precios] Verificando ${productos.length} productos`);

const resultados = [];
let hayProblemas = false;

for (const producto of productos) {
  try {
    // Buscar producto en master_list
    const { data: masterProducts } = await $supabase
      .from('master_list')
      .select('id, product_name')
      .ilike('product_name', `%${producto.produto}%`)
      .limit(1);
    
    if (!masterProducts || masterProducts.length === 0) {
      resultados.push({
        produto: producto.produto,
        problema: 'no_encontrado',
        mensaje: `Produto "${produto.produto}" não encontrado no catálogo`
      });
      hayProblemas = true;
      console.log(`⚠️ Produto no encontrado: ${produto.produto}`);
      continue;
    }
    
    const masterListId = masterProducts[0].id;
    
    // Verificar precios recientes (< 30 días)
    const fechaLimite = new Date();
    fechaLimite.setDate(fechaLimite.getDate() - parseInt($('Config Global').first().json.PRICE_VALIDITY_DAYS));
    
    const { data: recentPrices, error } = await $supabase
      .from('pricing_history')
      .select('id, supplier_id, unit_price, effective_date, suppliers(company_name)')
      .eq('master_list_id', masterListId)
      .gte('effective_date', fechaLimite.toISOString())
      .order('effective_date', { ascending: false })
      .limit(5);
    
    if (!recentPrices || recentPrices.length === 0) {
      // Buscar el precio más reciente aunque sea antiguo
      const { data: oldestPrice } = await $supabase
        .from('pricing_history')
        .select('effective_date, suppliers(company_name)')
        .eq('master_list_id', masterListId)
        .order('effective_date', { ascending: false })
        .limit(1)
        .single();
      
      if (oldestPrice) {
        const diasAtras = Math.floor((Date.now() - new Date(oldestPrice.effective_date)) / (1000 * 60 * 60 * 24));
        resultados.push({
          produto: produto.produto,
          problema: 'precos_desatualizados',
          dias_atras: diasAtras,
          ultimo_fornecedor: oldestPrice.suppliers?.company_name,
          mensaje: `Preços desatualizados (há ${diasAtras} dias)`
        });
        hayProblemas = true;
        console.log(`⚠️ Preços antigos para ${produto.produto}: ${diasAtras} días`);
      } else {
        resultados.push({
          produto: produto.produto,
          problema: 'sin_precios',
          mensaje: 'Sem preços cadastrados'
        });
        hayProblemas = true;
        console.log(`⚠️ Sin precios: ${produto.produto}`);
      }
    } else {
      resultados.push({
        produto: produto.produto,
        problema: null,
        precos_encontrados: recentPrices.length,
        mensaje: `✅ ${recentPrices.length} preços atualizados`
      });
      console.log(`✅ Preços OK para ${produto.produto}: ${recentPrices.length} opciones`);
    }
    
  } catch (error) {
    console.error(`❌ Error validando ${producto.produto}:`, error);
    resultados.push({
      produto: producto.produto,
      problema: 'erro',
      mensaje: 'Erro ao verificar preços'
    });
    hayProblemas = true;
  }
}

console.log(`🔍 [Validar Precios] Problemas detectados: ${hayProblemas}`);

return [{
  json: {
    phone_number: phoneNumber,
    restaurant_id: restaurantId,
    produtos: productos,
    validacion: resultados,
    hay_problemas: hayProblemas,
    puede_continuar: true  // Opción B: siempre permitir
  }
}];

const productos = $input.first().json.produtos;
const restaurantId = $input.first().json.restaurant_id;

console.log(`🔍 [Buscar Precios] Buscando precios para ${productos.length} productos`);

// Obtener preferencias del restaurante
const { data: restaurant } = await $supabase
  .from('restaurants')
  .select('price_sensitivity, preferred_suppliers')
  .eq('id', restaurantId)
  .single();

const priceSensitivity = restaurant?.price_sensitivity || 0.5;
const preferredSuppliers = restaurant?.preferred_suppliers || [];

console.log(`💡 [Buscar Precios] Price sensitivity: ${priceSensitivity}`);

const resultados = [];

for (const producto of productos) {
  try {
    // Buscar en master_list por similaridad
    const { data: masterProducts } = await $supabase
      .from('master_list')
      .select('id, product_name')
      .ilike('product_name', `%${producto.produto}%`)
      .limit(3);
    
    if (!masterProducts || masterProducts.length === 0) {
      console.log(`⚠️ Producto no encontrado en catálogo: ${producto.produto}`);
      resultados.push({
        produto: producto.produto,
        quantidade: produto.quantidade,
        unidade: produto.unidade,
        opciones: [],
        no_encontrado: true
      });
      continue;
    }
    
    const masterListIds = masterProducts.map(p => p.id);
    
    // Buscar TODOS los precios de TODOS los proveedores
    const { data: supplierPrices } = await $supabase
      .from('supplier_mapped_products')
      .select(\`
        id,
        supplier_id,
        supplier_product_name,
        current_unit_price,
        price_per_unit_type,
        suppliers (
          id,
          company_name,
          is_preferred
        )
      \`)
      .in('master_list_id', masterListIds)
      .eq('is_active', true);
    
    if (!supplierPrices || supplierPrices.length === 0) {
      console.log(`⚠️ No hay precios para: ${producto.produto}`);
      resultados.push({
        produto: producto.produto,
        quantidade: produto.quantidade,
        unidade: produto.unidade,
        opciones: [],
        sin_precios: true
      });
      continue;
    }
    
    console.log(`✅ Encontrados ${supplierPrices.length} precios para ${producto.produto}`);
    
    // Calcular score para cada opción
    const opciones = supplierPrices.map(price => {
      const supplier = price.suppliers;
      
      // Score base: precio (invertido y normalizado)
      const allPrices = supplierPrices.map(p => p.current_unit_price);
      const minPrice = Math.min(...allPrices);
      const maxPrice = Math.max(...allPrices);
      const priceRange = maxPrice - minPrice;
      
      let priceScore = 1.0;
      if (priceRange > 0) {
        priceScore = 1 - ((price.current_unit_price - minPrice) / priceRange);
      }
      
      // Bonus si es proveedor preferido
      const preferredBonus = preferredSuppliers.includes(supplier.id) ? 0.2 : 0;
      
      // Score final ponderado por price_sensitivity
      // price_sensitivity alto (0.8) = más importancia a calidad/preferido
      // price_sensitivity bajo (0.2) = más importancia a precio bajo
      const finalScore = (priceScore * (1 - priceSensitivity)) + 
                        (preferredBonus * priceSensitivity);
      
      return {
        supplier_id: supplier.id,
        supplier_name: supplier.company_name,
        product_name: price.supplier_product_name,
        price: price.current_unit_price,
        unit: price.price_per_unit_type,
        is_preferred: preferredSuppliers.includes(supplier.id),
        price_score: priceScore,
        final_score: finalScore
      };
    });
    
    // Ordenar por score
    opciones.sort((a, b) => b.final_score - a.final_score);
    
    resultados.push({
      produto: producto.produto,
      quantidade: produto.quantidade,
      unidade: produto.unidade,
      opciones: opciones,
      mejor_opcion: opciones[0],
      total_opciones: opciones.length
    });
    
  } catch (error) {
    console.error(`❌ Error buscando ${producto.produto}:`, error);
  }
}

console.log(`✅ [Buscar Precios] Análisis completo: ${resultados.length} productos`);

return [{
  json: {
    phone_number: $input.first().json.phone_number,
    restaurant_id: restaurantId,
    price_sensitivity: priceSensitivity,
    resultados: resultados,
    productos_analizados: resultados.length
  }
}];

// 🔄 Procesar cada precio y guardarlo
const pricesData = $input.first().json.prices_data;
const restaurantId = $('Preparar Datos Subir Precios').first().json.restaurant_id;

console.log('💾 [Procesar Precios] Procesando', pricesData.length, 'precios');

const results = [];

for (const item of pricesData) {
  try {
    // 1. Buscar o crear proveedor
    let supplierId = null;
    
    const { data: existingSupplier } = await $supabase
      .from('suppliers')
      .select('id')
      .ilike('company_name', item.fornecedor)
      .limit(1)
      .single();
    
    if (existingSupplier) {
      supplierId = existingSupplier.id;
      console.log('✅ Proveedor existente:', item.fornecedor, supplierId);
    } else {
      const { data: newSupplier, error } = await $supabase
        .from('suppliers')
        .insert({
          company_name: item.fornecedor,
          is_active: true,
          created_at: new Date().toISOString()
        })
        .select('id')
        .single();
      
      if (!error && newSupplier) {
        supplierId = newSupplier.id;
        console.log('✅ Nuevo proveedor creado:', item.fornecedor, supplierId);
      }
    }
    
    if (!supplierId) {
      console.error('❌ No se pudo obtener supplier_id para:', item.fornecedor);
      continue;
    }
    
    // 2. Buscar producto en master_list
    const { data: masterProduct } = await $supabase
      .from('master_list')
      .select('id')
      .ilike('product_name', `%${item.producto}%`)
      .limit(1)
      .single();
    
    const masterListId = masterProduct?.id || null;
    
    // 3. Guardar en supplier_mapped_products
    const { data: mappedProduct, error: mapError } = await $supabase
      .from('supplier_mapped_products')
      .insert({
        supplier_id: supplierId,
        master_list_id: masterListId,
        supplier_product_name: item.produto,
        current_unit_price: item.preco,
        price_per_unit_type: item.unidade,
        currency: 'BRL',
        mapping_confidence: masterListId ? 0.9 : 0.5,
        mapping_method: 'user_input',
        is_active: true,
        created_at: new Date().toISOString()
      })
      .select('id')
      .single();
    
    if (mapError) {
      console.error('❌ Error guardando producto:', mapError);
      continue;
    }
    
    // 4. Guardar en pricing_history
    await $supabase
      .from('pricing_history')
      .insert({
        supplier_id: supplierId,
        master_list_id: masterListId,
        supplier_mapped_product_id: mappedProduct.id,
        unit_price: item.preco,
        currency: 'BRL',
        price_per_unit_type: item.unidade,
        effective_date: new Date().toISOString(),
        snapshot_date: new Date().toISOString(),
        data_source: 'whatsapp_user',
        verification_status: 'pending',
        created_at: new Date().toISOString()
      });
    
    results.push({
      producto: item.produto,
      fornecedor: item.fornecedor,
      success: true
    });
    
    console.log('✅ Precio guardado:', item.produto, item.fornecedor, item.preco);
    
  } catch (error) {
    console.error('❌ Error procesando precio:', error);
    results.push({
      producto: item.produto,
      error: error.message
    });
  }
}

console.log(`✅ [Procesar Precios] ${results.filter(r => r.success).length}/${results.length} precios guardados`);

  json: {
    phone_number: $input.first().json.phone_number,

        currency: 'BRL',
        mapping_confidence: masterListId ? 0.8 : 0.5,
        mapping_method: 'user_input',
        is_active: true,
        created_at: new Date().toISOString()
      });
    
    console.log(`✅ Produto vinculado: ${producto}`);
  }
  
  return [{

    }
  }];

    json: {
      output: `❌ Erro ao cadastrar fornecedor. Tente novamente.`,
      phone_number: phoneNumber,
      error: error.message
    }
  }];

const usuario = $input.first().json;

console.log(`🔍 [Continuação] Mensaje: "${mensaje}"`);

// Opción 2: Atualizar preços

    mensaje.includes('preço') ||

      continuation_action: 'ATUALIZAR_PRECOS',

    }
  }];

// Opción 3: Registrar fornecedor
if (mensaje.includes('3') ||

      continuation_action: 'REGISTRAR_FORNECEDOR',

if (mensaje.includes('4') ||

  console.log('✅ [Continuação] Detectado: MOSTRAR_MENU');

  await $supabase
